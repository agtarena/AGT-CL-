create table if not exists public.agt_wallet_transactions ( id bigint generated always as identity primary key, user_id uuid not null references auth.users(id) on delete cascade, amount bigint not null, balance_after bigint not null, transaction_type text not null check (transaction_type in ('admin_credit','admin_debit','item_purchase','coin_purchase','refund')), description text not null default '', created_at timestamptz not null default now() ); create index if not exists agt_wallet_transactions_user_created_idx on public.agt_wallet_transactions(user_id, created_at desc); create table if not exists public.agt_store_items ( id uuid primary key default gen_random_uuid(), slug text not null unique, name text not null, category text not null check (category in ('frame','background','nick')), price integer not null check (price >= 0), description text not null default '', visual_key text not null default 'violet', active boolean not null default true, created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); create table if not exists public.agt_user_items ( user_id uuid not null references auth.users(id) on delete cascade, item_id uuid not null references public.agt_store_items(id) on delete cascade, purchased_at timestamptz not null default now(), primary key(user_id, item_id) ); create table if not exists public.agt_profile_equipment ( user_id uuid not null references auth.users(id) on delete cascade, category text not null check (category in ('frame','background','nick')), item_id uuid references public.agt_store_items(id) on delete set null, equipped_at timestamptz not null default now(), primary key(user_id, category) );
